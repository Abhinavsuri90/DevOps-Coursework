# INVESTIGATE: describe shows sizes without revealing values - the client's copy is one byte longer
$ kubectl describe secret db-admin ledger-api-db | grep -E '^Name:|password:'

Name:         ledger-api-db
password:  25 bytes
# Inspect only the LAST byte of each value (the password itself is never printed)
$ kubectl get secret ledger-api-db -o jsonpath='{.data.password}' | base64 --decode | tail -c 1 | xxd
00000000: 0a                                       .
$ kubectl get secret db-admin -o jsonpath='{.data.password}' | base64 --decode | tail -c 1 | xxd -p | sed 's/../last byte is a hex digit (0x&), not a newline/'
last byte is a hex digit (0x37), not a newline
# Why: echo appends a newline before base64; printf %s does not
$ echo 'mypassword' | base64; printf %s 'mypassword' | base64
bXlwYXNzd29yZAo=
bXlwYXNzd29yZA==
# ROOT CAUSE: ledger-api-db was built with echo, so PGPASSWORD ends in \n and no longer matches POSTGRES_PASSWORD
