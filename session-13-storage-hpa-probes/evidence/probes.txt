# 1) Readiness probe gates traffic: Running but not Ready, so the Service will not route to it (client: busybox Pod probe-client)
$ kubectl apply -f session-13-storage-hpa-probes/03-probes/readiness-gate.yaml
deployment.apps/probe-readiness created
service/probe-readiness created
$ kubectl get pods -l app=probe-readiness
NAME                               READY   STATUS    RESTARTS   AGE
probe-readiness-78f7dd88fb-6fbwd   0/1     Running   0          15s
$ kubectl get endpointslice -l kubernetes.io/service-name=probe-readiness -o jsonpath='{range .items[*].endpoints[*]}{.addresses[0]}  ready={.conditions.ready}{"\n"}{end}'
10.244.0.79  ready=false
$ kubectl describe pod -l app=probe-readiness | sed -n '/^Events:/,$p' | grep Unhealthy | tail -1 | cut -c1-160
  Warning  Unhealthy  2s (x5 over 14s)  kubelet            spec.containers{nginx}: Readiness probe failed: HTTP probe failed with statuscode: 404
$ kubectl exec probe-client -- wget -qO- -T 3 http://probe-readiness
wget: can't connect to remote host (10.96.193.11): Connection refused
command terminated with exit code 1
# Fix: point the readiness probe at a path that exists
$ kubectl patch deployment probe-readiness --type json -p '[{"op":"replace","path":"/spec/template/spec/containers/0/readinessProbe/httpGet/path","value":"/"}]'
deployment.apps/probe-readiness patched
$ kubectl rollout status deployment/probe-readiness --timeout=120s
Waiting for deployment "probe-readiness" rollout to finish: 1 old replicas are pending termination...
Waiting for deployment "probe-readiness" rollout to finish: 1 old replicas are pending termination...
Waiting for deployment "probe-readiness" rollout to finish: 1 old replicas are pending termination...
deployment "probe-readiness" successfully rolled out
$ kubectl get pods -l app=probe-readiness
NAME                               READY   STATUS    RESTARTS   AGE
probe-readiness-58d79d654f-jvhjr   1/1     Running   0          4s
$ kubectl get endpointslice -l kubernetes.io/service-name=probe-readiness -o jsonpath='{range .items[*].endpoints[*]}{.addresses[0]}  ready={.conditions.ready}{"\n"}{end}'

$ kubectl delete -f session-13-storage-hpa-probes/03-probes/readiness-gate.yaml

# 2) Liveness probe restarts an unhealthy container
$ kubectl apply -f session-13-storage-hpa-probes/03-probes/liveness-restart.yaml
pod/probe-liveness created
$ kubectl get pod probe-liveness
NAME             READY   STATUS    RESTARTS     AGE
probe-liveness   1/1     Running   4 (1s ago)   40s
$ kubectl describe pod probe-liveness | sed -n '/^Events:/,$p' | grep -E 'Unhealthy|Killing' | tail -2 | cut -c1-170
  Warning  Unhealthy  1s (x8 over 37s)   kubelet            spec.containers{nginx}: Liveness probe failed: HTTP probe failed with statuscode: 404
  Normal   Killing    1s (x4 over 34s)   kubelet            spec.containers{nginx}: Container nginx failed liveness probe, will be restarted
$ kubectl delete pod probe-liveness --grace-period=1
pod "probe-liveness" deleted from default namespace
