# 4. Argo CD notices the commit on main and rolls out the new images
$ SECONDS=0; until [ "$(kubectl -n argocd get applications.argoproj.io spendwise -o jsonpath='{.status.sync.revision}')" = "$(git -C .. rev-parse origin/main)" ] && kubectl -n argocd get applications.argoproj.io spendwise -o jsonpath='{.status.sync.status}{.status.health.status}' | grep -q SyncedHealthy; do sleep 5; done; echo "Argo CD synced and Healthy ${SECONDS}s after the GitOps commit appeared"
