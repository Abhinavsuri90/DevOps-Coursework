# Named Service ports also publish SRV records: _<port-name>._<protocol>.<service>.<namespace>.svc.cluster.local
$ kubectl -n shop exec netclient -c dns -- dig +short SRV _http._tcp.billing.payments.svc.cluster.local
0 100 80 billing.payments.svc.cluster.local.
# Pod A record: dashed Pod IP + namespace + pod.cluster.local
