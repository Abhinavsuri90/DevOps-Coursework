# Self-heal: someone changes the live cluster by hand (drift). Git still says replicas: 3.
$ kubectl -n gitops-demo scale deployment gitops-web --replicas=1
deployment.apps/gitops-web scaled
$ kubectl -n gitops-demo get deploy gitops-web -o jsonpath='spec.replicas right after the manual change: {.spec.replicas}{"\n"}'
spec.replicas right after the manual change: 1
$ SECONDS=0; until [ "$(kubectl -n gitops-demo get deploy gitops-web -o jsonpath='{.spec.replicas}')" = 3 ]; do sleep 1; done; echo "Argo CD reverted the drift after ${SECONDS}s"
Argo CD reverted the drift after 2s
